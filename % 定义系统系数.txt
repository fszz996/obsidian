clc;
clear;

f = [1 2 3 4 5];

% 验证 f[n]*delta[n] = f[n]
delta = 1;
y1 = conv(f,delta);
e1 = y1 - f;

% 验证 f[n]*delta[n-3] = f[n-3]
delta0 = [0 0 0 1];
y2 = conv(f,delta0);
f_delay = [zeros(1,3) f];
e2 = y2 - f_delay;

figure;

subplot(3,2,1);
stem(0:length(y1)-1,y1);
title('f[n]*delta[n]');
grid on;

subplot(3,2,2);
stem(0:length(f)-1,f);
title('f[n]');
grid on;

subplot(3,2,3);
stem(0:length(y2)-1,y2);
title('f[n]*delta[n-3]');
grid on;

subplot(3,2,4);
stem(0:length(f_delay)-1,f_delay);
title('f[n-3]');
grid on;

subplot(3,2,5);
stem(0:length(e1)-1,e1);
title('第一组误差');
grid on;

subplot(3,2,6);
stem(0:length(e2)-1,e2);
title('第二组误差');
grid on;

clc;

f1 = [1 2 3 4];
f2 = [2 3 1];

% 左边
y1 = conv(f1,f2);

% 右边
y2 = conv(f2,f1);

% 误差
e = y1-y2;

figure;

subplot(3,1,1);
stem(0:length(y1)-1,y1);
title('f1[n]*f2[n]');
grid on;

subplot(3,1,2);
stem(0:length(y2)-1,y2);
title('f2[n]*f1[n]');
grid on;

subplot(3,1,3);
stem(0:length(e)-1,e);
title('卷积交换律误差');
xlabel('n');
ylabel('误差');
grid on;

clc;
clear;

f1 = [1 2 3];
f2 = [2 3 4];
f3 = [1 3 2];

% 左边：先相加，再卷积
y1 = conv(f1,f2+f3);

% 右边：分别卷积，再相加
y2 = conv(f1,f2)+conv(f1,f3);

% 误差
e = y1-y2;

figure;

subplot(3,1,1);
stem(0:length(y1)-1,y1);
title('f1[n]*(f2[n]+f3[n])');
grid on;

subplot(3,1,2);
stem(0:length(y2)-1,y2);
title('f1[n]*f2[n]+f1[n]*f3[n]');
grid on;

subplot(3,1,3);
stem(0:length(e)-1,e);
title('卷积分配律误差');
xlabel('n');
ylabel('误差');
grid on;
